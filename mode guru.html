Panduan Tabel SQL Supabase (Jika Belum Dibuat)

Salin skrip SQL di bawah ini dan jalankan pada menu SQL Editor di dashboard Supabase Anda agar penyimpanan data presensi dan akun guru berjalan sempurna:

-- Tabel Akun Guru
CREATE TABLE IF NOT EXISTS guru (
    id SERIAL PRIMARY KEY,
    nama TEXT NOT NULL,
    nip TEXT UNIQUE NOT NULL,
    tugas TEXT,
    created_at TIMESTAMP DEFAULT NOW()
);

-- Tabel Presensi / Absensi Siswa
CREATE TABLE IF NOT EXISTS presensi (
    id SERIAL PRIMARY KEY,
    nis TEXT,
    nama TEXT,
    kelas TEXT,
    status TEXT,
    tanggal DATE DEFAULT CURRENT_DATE,
    created_at TIMESTAMP DEFAULT NOW()
);
